% Notation for the computational leg (the two-hop ladder).
% A public key; \pk_B and \pk_C are Bob's and Charlie's.
\newcommand{\pk}{\mathsf{pk}}
% Encryption under a public key.
\newcommand{\Enc}{\mathsf{Enc}}
% The real-or-zero advantage at a public key of a single-query adversary:
% \epsror{pk}{A} is the gap between A's acceptance probability on an encryption
% of the plaintext it chose and on an encryption of zero.
\newcommand{\epsror}[2]{\varepsilon_{\mathrm{ror}}(#1,\,#2)}
% The hop-i reduction a distinguisher induces on the i-th received ciphertext.
\newcommand{\hopred}[2]{\mathcal{R}_{#1}(#2)}
% The sum of the advantages of a distinguisher's two hop reductions, at the key
% of the party that sent each of the two ciphertexts Alice receives.
\newcommand{\epshopsum}[1]{\epsror{\pk_B}{\hopred{0}{#1}} + \epsror{\pk_C}{\hopred{1}{#1}}}
% Alice's hopping tuple at hop i: her masks, her combine randomness, the output,
% and the two received ciphertexts with the first i of them encrypting zero.
\newcommand{\hoptuple}[1]{T_{#1}}
