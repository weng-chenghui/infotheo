    Definition PCM r : 'M_(r, n):= Vandermonde r a *m diag_mx b.

    Definition syndrome_coord (i : nat) (y : 'rV_n) :=

    Definition syndrome r (y : 'rV_n) : 'rV[F]_r := \row_i (syndrome_coord i y).
